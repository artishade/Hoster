     7→/**
     8→ * Real service runtime.
     9→ *
    10→ * Every service in "running" state gets a REAL HTTP server bound to a real
    11→ * 127.0.0.1 port inside the control-plane host. Requests reach it through the
    12→ * same-origin ingress proxy (/api/ingress/<name>) so deployed projects are
    13→ * actually visitable AND interactive — no DNS records required.
    14→ *
    15→ * Each runner serves:
    16→ *   - a live interactive console at / (API playground, persistent KV data
    17→ *     store, real-time request feed, SSE stream for streaming services)
    18→ *   - real JSON endpoints: /health, /api/meta, /api/stats, /api/time,
    19→ *     /api/echo, /api/kv[/:key], /api/requests, /sse
    20→ *   - genuine persistent state (SQLite-backed ServiceData KV store) and
    21→ *     genuine request metrics that survive restarts
    22→ *
    23→ * The pretty hostname (https://<name>.nexushost.dev) remains the "custom
    24→ * domain" view: it only resolves once a wildcard DNS record points at this
    25→ * server, exactly like any real platform. The UI marks it as DNS-pending.
    26→ */

    71→function safeParse<T>(json: string | null | undefined, fallback: T): T {
    72→  if (!json) return fallback;
    73→  try {
    74→    return JSON.parse(json) as T;
    75→  } catch {
    76→    return fallback;
    77→  }

    99→function isPortFree(port: number): Promise<boolean> {
   100→  return new Promise((resolve) => {
   101→    const probe = net.createServer();
   102→    probe.once('error', () => resolve(false));
   103→    probe.once('listening', () => probe.close(() => resolve(true)));
   104→    probe.listen(port, '127.0.0.1');
   105→  });
   106→}

   108→function portForService(id: string): number {
   109→  const hash = crypto.createHash('sha1').update(id).digest();
   110→  return PORT_BASE + (hash.readUInt16BE(0) % PORT_RANGE);
   111→}
   112→
   113→async function allocatePort(id: string): Promise<number> {
   114→  const preferred = portForService(id);
   115→  for (let i = 0; i < MAX_PORT_TRIES; i++) {
   116→    const candidate = PORT_BASE + (((preferred - PORT_BASE + i) % PORT_RANGE) + PORT_RANGE) % PORT_RANGE;
   117→    if (await isPortFree(candidate)) return candidate;
   118→  }
   119→  return preferred; // let listen() surface the error if truly exhausted
   120→}

   140→function emptyStats(): LiveState['stats'] {
   141→  return { totalRequests: 0, okResponses: 0, errorResponses: 0, totalMs: 0, byPath: {}, lastRequestAt: null };

   144→function getLiveState(serviceId: string): LiveState {
   145→  let st = liveStates.get(serviceId);
   146→  if (!st) {
   147→    st = { stats: emptyStats(), ring: [], latencies: [], flushTimer: null };
   148→    liveStates.set(serviceId, st);
   149→  }
   150→  return st;
   151→}

   153→async function loadPersistedStats(serviceId: string): Promise<void> {
   154→  try {
   155→    const row = await db.serviceData.findUnique({
   156→      where: { serviceId_key: { serviceId, key: STATS_KEY } },
   157→      select: { valueJson: true },
   158→    });
   159→    if (!row) return;
   160→    const saved = safeParse<Partial<LiveState['stats']> | null>(row.valueJson, null);
   161→    if (saved && typeof saved.totalRequests === 'number') {
   162→      const st = getLiveState(serviceId);
   163→      st.stats = { ...emptyStats(), ...saved, byPath: saved.byPath ?? {} };
   164→    }
   165→  } catch {
   166→    /* stats are best-effort */
   167→  }
   168→}

   170→function scheduleStatsFlush(serviceId: string): void {
   171→  const st = getLiveState(serviceId);
   172→  if (st.flushTimer) return;
   173→  st.flushTimer = setTimeout(() => {
   174→    st.flushTimer = null;
   175→    const snapshot = JSON.stringify(st.stats);
   176→    db.serviceData
   177→      .upsert({
   178→        where: { serviceId_key: { serviceId, key: STATS_KEY } },
   179→        update: { valueJson: snapshot },
   180→        create: { serviceId, key: STATS_KEY, valueJson: snapshot },
   181→      })
   182→      .catch(() => {});
   183→  }, 2500);
   184→  st.flushTimer.unref?.();
   185→}

   187→function recordRequest(serviceId: string, entry: { method: string; path: string; status: number; ms: number }): void {
   188→  const st = getLiveState(serviceId);
   189→  st.stats.totalRequests += 1;
   190→  if (entry.status < 400) st.stats.okResponses += 1;
   191→  else st.stats.errorResponses += 1;
   192→  st.stats.totalMs += entry.ms;
   193→  st.stats.lastRequestAt = new Date().toISOString();
   194→
   195→  const bucket = st.stats.byPath[entry.path] ?? { count: 0, totalMs: 0 };
   196→  bucket.count += 1;
   197→  bucket.totalMs += entry.ms;
   198→  st.stats.byPath[entry.path] = bucket;
   199→
   200→  st.ring.push({ ...entry, at: new Date().toISOString() });
   201→  if (st.ring.length > RING_CAP) st.ring.splice(0, st.ring.length - RING_CAP);
   202→
   203→  st.latencies.push(entry.ms);
   204→  if (st.latencies.length > LATENCY_CAP) st.latencies.splice(0, st.latencies.length - LATENCY_CAP);
   205→
   206→  scheduleStatsFlush(serviceId);
   207→}

   220→export function liveRequestsLastMinute(serviceId: string): number {
   221→  const st = getLiveState(serviceId);
   222→  const cutoff = Date.now() - 60_000;
   223→  return st.ring.filter((r) => new Date(r.at).getTime() >= cutoff).length;
   224→}

   226→export function liveLatencyP95(serviceId: string): number {
   227→  return percentile(getLiveState(serviceId).latencies, 95);

   230→export function liveLatencyAvg(serviceId: string): number {
   231→  const st = getLiveState(serviceId);
   232→  return st.stats.totalRequests ? Math.round(st.stats.totalMs / st.stats.totalRequests) : 0;
   233→}

   235→function percentile(values: number[], p: number): number {
   236→  if (!values.length) return 0;
   237→  const sorted = [...values].sort((a, b) => a - b);
   238→  const idx = Math.min(sorted.length - 1, Math.ceil((p / 100) * sorted.length) - 1);
   239→  return Math.round(sorted[Math.max(0, idx)]);
   240→}

   698→function readBody(req: http.IncomingMessage): Promise<Buffer> {
   699→  return new Promise((resolve, reject) => {
   700→    const chunks: Buffer[] = [];
   701→    req.on('data', (c: Buffer) => {
   702→      chunks.push(c);
   703→      if (chunks.reduce((n, b) => n + b.length, 0) > MAX_VALUE_BYTES + 4096) {
   704→        reject(new Error('payload too large'));
   705→        req.destroy();
   706→      }
   707→    });
   708→    req.on('end', () => resolve(Buffer.concat(chunks)));
   709→    req.on('error', reject);
   710→  });
   711→}

   719→function normalizeUrl(raw: string): string {
   720→  const q = raw.indexOf('?');
   721→  const path = q === -1 ? raw : raw.slice(0, q);
   722→  if (path === '/' || path === '') return '/';
   723→  return path.replace(/\/+$/, '') || '/';
   724→}

   726→function createRunnerApp(
   727→  serviceId: string,
   728→  name: string,
   729→  meta: RunnerMeta,
   730→  port: number
   731→): http.Server {
   732→  const streamEnabled = meta.protocol === 'sse' || meta.type === 'mcp';
   733→  const bootAt = Date.now();
   734→
   735→  async function route(req: http.IncomingMessage, res: http.ServerResponse): Promise<void> {
   736→    const url = normalizeUrl(req.url ?? '/');
   737→    const method = (req.method ?? 'GET').toUpperCase();
   738→
   739→    // CORS-friendly preflight (harmless same-origin, useful for direct API use)
   740→    if (method === 'OPTIONS') {
   741→      res.writeHead(204, {
   742→        'access-control-allow-origin': '*',
   743→        'access-control-allow-methods': 'GET, POST, PUT, PATCH, DELETE, OPTIONS',
   744→        'access-control-allow-headers': 'content-type',
   745→      });
   746→      res.end();
   747→      return;
   748→    }
   749→
   750→    if (url === '/health') {
   751→      sendJson(res, 200, { status: 'ok', service: name, uptimeSec: Math.round((Date.now() - bootAt) / 1000) });
   752→      return;
   753→    }
   754→
   755→    if (url === '/api/meta') {
   756→      sendJson(res, 200, {
   757→        service: name,
   758→        type: meta.type,
   759→        protocol: meta.protocol,
   760→        description: meta.description,
   761→        commit: meta.commit,
   762→        branch: meta.branch,
   763→        hardwareTier: meta.tier,
   764→        region: meta.region,
   765→        console: 'GET / opens the interactive console',
   766→        runtime: { mode: 'builtin-runner', version: RUNNER_VERSION },
   767→        endpoints: ['GET /health', 'GET /api/meta', 'GET /api/stats', 'GET /api/time', 'POST /api/echo', 'GET /api/kv', 'PUT /api/kv/{key}', 'DELETE /api/kv/{key}', 'GET /api/requests'].concat(streamEnabled ? ['GET /sse'] : []),
   768→      });
   769→      return;
   770→    }
   771→
   772→    if (url === '/api/stats') {
   773→      const st = getLiveState(serviceId);
   774→      const last60 = st.ring.filter((r) => Date.now() - new Date(r.at).getTime() < 60_000);
   775→      sendJson(res, 200, {
   776→        service: name,
   777→        startedAtMs: bootAt,
   778→        startedAt: new Date(bootAt).toISOString(),
   779→        uptimeSec: Math.round((Date.now() - bootAt) / 1000),
   780→        totalRequests: st.stats.totalRequests,
   781→        okResponses: st.stats.okResponses,
   782→        errorResponses: st.stats.errorResponses,
   783→        totalMs: Math.round(st.stats.totalMs),
   784→        avgLatencyMs: st.stats.totalRequests ? Math.round(st.stats.totalMs / st.stats.totalRequests) : 0,
   785→        p95LatencyMs: percentile(st.latencies, 95),
   786→        requestsLast60s: last60.length,
   787→        requestsPerMin: last60.length,
   788→        byPath: st.stats.byPath,
   789→        lastRequestAt: st.stats.lastRequestAt,
   790→        runtime: { pid: process.pid, port, mode: 'builtin-runner' },
   791→      });
   792→      return;
   793→    }
   794→
   795→    if (url === '/api/time') {
   796→      const now = new Date();
   797→      sendJson(res, 200, { iso: now.toISOString(), epochMs: now.getTime(), timezone: 'UTC', runnerUptimeSec: Math.round((Date.now() - bootAt) / 1000) });
   798→      return;
   799→    }
   800→
   801→    if (url === '/api/echo') {
   802→      let raw = '';
   803→      let payload: unknown = null;
   804→      try {
   805→        raw = (await readBody(req)).toString('utf8');
   806→      } catch {
   807→        sendJson(res, 413, { error: `Body too large (max ${Math.round(MAX_VALUE_BYTES / 1024)} KB)` });
   808→        return;
   809→      }
   810→      payload = raw ? safeParse<unknown>(raw, raw) : null;
   811→      sendJson(res, 200, {
   812→        service: name,
   813→        method,
   814→        receivedAt: new Date().toISOString(),
   815→        contentType: req.headers['content-type'] ?? null,
   816→        bodyBytes: raw.length,
   817→        body: payload,
   818→      });
   819→      return;
   820→    }
   821→
   822→    if (url === '/api/kv' && method === 'GET') {
   823→      const rows = await db.serviceData.findMany({
   824→        where: { serviceId },
   825→        orderBy: { updatedAt: 'desc' },
   826→      });
   827→      // NOTE: filter internal "__" keys in JS — SQLite LIKE treats "_" as a
   828→      // wildcard, so a NOT startsWith('__') clause would exclude everything.
   829→      sendJson(res, 200, {
   830→        data: rows
   831→          .filter((r) => !r.key.startsWith('__'))
   832→          .map((r) => ({ key: r.key, value: safeParse<unknown>(r.valueJson, r.valueJson), updatedAt: r.updatedAt })),
   833→      });
   834→      return;
   835→    }
   836→
   837→    if (url.startsWith('/api/kv/')) {
   838→      const key = decodeURIComponent(url.slice('/api/kv/'.length));
   839→      if (!key || key.length > MAX_KEY_LEN || key.startsWith('__')) {
   840→        sendJson(res, 400, { error: 'Invalid key (max 120 chars, no __ prefix)' });
   841→        return;
   842→      }
   843→
   844→      if (method === 'GET') {
   845→        const row = await db.serviceData.findUnique({ where: { serviceId_key: { serviceId, key } } });
   846→        if (!row) {
   847→          sendJson(res, 404, { error: `Key "${key}" not found`, service: name });
   848→          return;
   849→        }
   850→        sendJson(res, 200, { key: row.key, value: safeParse<unknown>(row.valueJson, row.valueJson), updatedAt: row.updatedAt });
   851→        return;
   852→      }
   853→
   854→      if (method === 'PUT' || method === 'POST' || method === 'PATCH') {
   855→        let raw = '';
   856→        try {
   857→          raw = (await readBody(req)).toString('utf8');
   858→        } catch {
   859→          sendJson(res, 413, { error: `Value too large (max ${Math.round(MAX_VALUE_BYTES / 1024)} KB)` });
   860→          return;
   861→        }
   862→        const row = await db.serviceData.upsert({
   863→          where: { serviceId_key: { serviceId, key } },
   864→          update: { valueJson: raw },
   865→          create: { serviceId, key, valueJson: raw },
   866→        });
   867→        sendJson(res, row.createdAt.getTime() === row.updatedAt.getTime() ? 201 : 200, {
   868→          key: row.key,
   869→          value: safeParse<unknown>(row.valueJson, row.valueJson),
   870→          updatedAt: row.updatedAt,
   871→          persisted: true,
   872→        });
   873→        return;
   874→      }
   875→
   876→      if (method === 'DELETE') {
   877→        try {
   878→          await db.serviceData.delete({ where: { serviceId_key: { serviceId, key } } });
   879→          sendJson(res, 200, { deleted: true, key });
   880→        } catch {
   881→          sendJson(res, 404, { error: `Key "${key}" not found`, service: name });
   882→        }
   883→        return;
   884→      }
   885→
   886→      sendJson(res, 405, { error: `Method ${method} not allowed on /api/kv/{key}` });
   887→      return;
   888→    }
   889→
   890→    if (url === '/api/requests') {
   891→      const st = getLiveState(serviceId);
   892→      sendJson(res, 200, { data: st.ring.slice(-RING_CAP), total: st.stats.totalRequests });
   893→      return;
   894→    }
   895→
   896→    if (url === '/sse' && streamEnabled) {
   897→      res.writeHead(200, {
   898→        'content-type': 'text/event-stream',
   899→        'cache-control': 'no-cache',
   900→        connection: 'keep-alive',
   901→      });
   902→      let tick = 0;
   903→      const timer = setInterval(() => {
   904→        tick += 1;
   905→        res.write(`event: heartbeat\ndata: ${JSON.stringify({ service: name, tick, at: new Date().toISOString(), uptimeSec: Math.round((Date.now() - bootAt) / 1000) })}\n\n`);
   906→        if (tick >= 600) { clearInterval(timer); res.end(); }
   907→      }, 1000);
   908→      req.on('close', () => clearInterval(timer));
   909→      return;
   910→    }
   911→
   912→    if (url === '/' || url === '') {
   913→      res.writeHead(200, { 'content-type': 'text/html; charset=utf-8', 'cache-control': 'no-store' });
   914→      res.end(renderAppPage(name, meta, streamEnabled));
   915→      return;
   916→    }
   917→
   918→    sendJson(res, 404, { error: 'not found', service: name, path: url, hint: 'GET / opens the interactive console — GET /api/meta lists endpoints' });
   919→  }
   920→
   921→  const server = http.createServer((req, res) => {
   922→    const t0 = Date.now();
   923→    const rawPath = (req.url ?? '/').split('?')[0] || '/';
   924→    const isStream = rawPath === '/sse';
   925→    if (!isStream) {
   926→      res.on('finish', () => {
   927→        try {
   928→          recordRequest(serviceId, { method: (req.method ?? 'GET').toUpperCase(), path: rawPath.slice(0, 120), status: res.statusCode, ms: Date.now() - t0 });
   929→        } catch {
   930→          /* metrics must never break serving */
   931→        }
   932→      });
   933→    }
   934→    route(req, res).catch((err) => {
   935→      console.error(`[runner:${name}] handler error`, err);
   936→      if (!res.headersSent) sendJson(res, 500, { error: 'internal runner error', service: name });
   937→      else res.end();
   938→    });
   939→  });
   940→  (server as http.Server & { __v?: number }).__v = RUNNER_VERSION;
   941→  return server;
   942→}

   946→async function startRuntime(svc: {
   947→  id: string; name: string; type: string; commitHash: string; branch: string;
   948→  hardwareTier: string; region: string; description: string; protocol: string;
   949→}): Promise<ServiceRuntime> {
   950→  const meta: RunnerMeta = {
   951→    type: svc.type,
   952→    commit: svc.commitHash,
   953→    branch: svc.branch,
   954→    tier: svc.hardwareTier,
   955→    region: svc.region,
   956→    description: svc.description,
   957→    protocol: svc.protocol,
   958→  };
   959→
   960→  const port = await allocatePort(svc.id);
   961→  const server = createRunnerApp(svc.id, svc.name, meta, port);
   962→
   963→  await new Promise<void>((resolve, reject) => {
   964→    server.once('error', reject);
   965→    server.listen(port, '127.0.0.1', () => resolve());
   966→  });
   967→  runners.set(svc.id, server);
   968→
   969→  const state: RuntimeState = {
   970→    pid: process.pid,
   971→    port,
   972→    startedAt: new Date().toISOString(),
   973→    mode: 'builtin-runner',
   974→    healthy: true,
   975→  };
   976→  await db.service.update({ where: { id: svc.id }, data: { runtimeJson: JSON.stringify(state) } });
   977→  await loadPersistedStats(svc.id);
   978→  await writeLog(svc.id, `Runtime for "${svc.name}" is listening on 127.0.0.1:${port} (builtin runner v${RUNNER_VERSION}). Interactive console: /api/ingress/${svc.name}`);
   979→  return { ...state, healthy: true };
   980→}

   982→export async function stopRuntime(svc: { id: string; name: string }): Promise<void> {
   983→  const server = runners.get(svc.id);
   984→  if (server) {
   985→    (server as http.Server & { closeIdleConnections?: () => void }).closeIdleConnections?.();
   986→    await new Promise<void>((resolve) => server.close(() => resolve()));
   987→    runners.delete(svc.id);
   988→    await writeLog(svc.id, `Runtime for "${svc.name}" terminated — port released.`);
   989→  }
   990→  const row = await db.service.findUnique({ where: { id: svc.id }, select: { runtimeJson: true } });
   991→  if (row?.runtimeJson) {
   992→    await db.service.update({ where: { id: svc.id }, data: { runtimeJson: null } });
   993→  }
   994→}

  1001→/** Replace a stale runner (old app version / unhealthy) with a fresh one. */
  1002→async function replaceRuntime(svc: {
  1003→  id: string; name: string; status: string; type: string; commitHash: string; branch: string;
  1004→  hardwareTier: string; region: string; description: string; protocol: string;
  1005→}): Promise<ServiceRuntime | null> {
  1006→  const old = runners.get(svc.id);
  1007→  if (old) {
  1008→    await new Promise<void>((resolve) => old.close(() => resolve()));
  1009→    runners.delete(svc.id);
  1010→  }
  1011→  return startRuntime(svc);
  1012→}

  1014→/**
  1015→ * Reconcile one service with the desired state:
  1016→ *  - running  → guarantee a live listener with the CURRENT app version
  1017→ *  - anything else → guarantee no listener
  1018→ * Returns fresh runtime info for running services.
  1019→ *
  1020→ * Git-deployed services (repoUrl set) are managed by deployer.ts — their
  1021→ * runtimeJson.mode is 'git-deploy' and they own a REAL child process. We only
  1022→ * trust + health-check them here, never replacing them with a builtin runner.
  1023→ */
  1024→export async function ensureRuntime(svc: {
  1025→  id: string; name: string; status: string; type: string; commitHash: string; branch: string;
  1026→  hardwareTier: string; region: string; description: string; protocol: string;
  1027→  repoUrl?: string;
  1028→}): Promise<ServiceRuntime | null> {
  1029→  if (svc.status !== 'running') {
  1030→    await stopRuntime(svc);
  1031→    return null;
  1032→  }
  1033→
  1034→  // ── git-deployed service: trust the deployer's real child process ──────
  1035→  const stored = await readRuntimeState(svc.id);
  1036→  if (stored?.mode === 'git-deploy') {
  1037→    const healthy = await isHealthy(stored.port);
  1038→    if (healthy !== stored.healthy) {
  1039→      const updated = { ...stored, healthy };
  1040→      await db.service.update({ where: { id: svc.id }, data: { runtimeJson: JSON.stringify(updated) } });
  1041→      return updated;
  1042→    }
  1043→    return { ...stored, healthy };
  1044→  }
  1045→
  1046→  const inProcess = runners.get(svc.id);
  1047→  if (inProcess) {
  1048→    const version = (inProcess as http.Server & { __v?: number }).__v;
  1049→    if (version !== RUNNER_VERSION) {
  1050→      return replaceRuntime(svc);
  1051→    }
  1052→    if (stored) {
  1053→      const healthy = await isHealthy(stored.port);
  1054→      if (healthy !== stored.healthy) {
  1055→        const updated = { ...stored, healthy };
  1056→        await db.service.update({ where: { id: svc.id }, data: { runtimeJson: JSON.stringify(updated) } });
  1057→        return updated;
  1058→      }
  1059→      return { ...stored, healthy };
  1060→    }
  1061→  }
  1062→
  1063→  // Not served by THIS process — maybe another worker/instance already serves it.
  1064→  if (stored) {
  1065→    const healthy = await isHealthy(stored.port);
  1066→    if (healthy) {
  1067→      if (!stored.healthy) {
  1068→        const updated = { ...stored, healthy: true };
  1069→        await db.service.update({ where: { id: svc.id }, data: { runtimeJson: JSON.stringify(updated) } });
  1070→        return updated;
  1071→      }
  1072→      return { ...stored, healthy: true };
  1073→    }
  1074→  }
  1075→
  1076→  const fresh = await startRuntime(svc);
  1077→  return fresh;
  1078→}

  1080→type ReconcileGlobal = typeof globalThis & { __hosterReconcile?: { busy: boolean; lastRun: number } };
  1081→const rg = globalThis as ReconcileGlobal;

  1083→/** Reconciles all services (started missing runners, stops orphans). Cheap to call often. */
  1084→export async function reconcileRuntimes(): Promise<void> {
  1085→  rg.__hosterReconcile ??= { busy: false, lastRun: 0 };
  1086→  const rec = rg.__hosterReconcile;
  1087→  if (rec.busy || Date.now() - rec.lastRun < 2_000) return;
  1088→  rec.busy = true;
  1089→  try {
  1090→    const rows = await db.service.findMany({
  1091→      where: { OR: [{ status: 'running' }, { runtimeJson: { not: null } }] },
  1092→    });
  1093→    for (const row of rows) {
  1094→      // git-deployed services run as real child processes owned by deployer.ts;
  1095→      // ensureRuntime health-checks them but never spawns a builtin runner.
  1096→      try {
  1097→        await ensureRuntime(row);
  1098→      } catch (err) {
  1099→        console.error(`[runtime] reconcile failed for ${row.name}`, err);
  1100→      }
  1101→    }
  1102→  } finally {
  1103→    rec.busy = false;
  1104→    rec.lastRun = Date.now();
  1105→  }
  1106→}
