    90→    90→function isPortFree(port: number): Promise<boolean> {
    91→    91→  return new Promise((resolve) => {
    92→    92→    const probe = net.createServer();
    93→    93→    probe.once('error', () => resolve(false));
    94→    94→    probe.once('listening', () => probe.close(() => resolve(true)));
    95→    95→    probe.listen(port, '127.0.0.1');
    96→    96→  });
    97→    97→}

    99→    99→function portForService(id: string): number {
   100→   100→  const hash = crypto.createHash('sha1').update(id).digest();
   101→   101→  return PORT_BASE + (hash.readUInt16BE(0) % PORT_RANGE);
   102→   102→}
   103→   103→
   104→   104→async function allocatePort(id: string): Promise<number> {
   105→   105→  const preferred = portForService(id);
   106→   106→  for (let i = 0; i < MAX_PORT_TRIES; i++) {
   107→   107→    const candidate = PORT_BASE + (((preferred - PORT_BASE + i) % PORT_RANGE) + PORT_RANGE) % PORT_RANGE;
   108→   108→    if (await isPortFree(candidate)) return candidate;
   109→   109→  }
   110→   110→  return preferred; // let listen() surface the error if truly exhausted
   111→   111→}

   143→   143→async function loadPersistedStats(serviceId: string): Promise<void> {
   144→   144→  try {
   145→   145→    const row = await db.serviceData.findUnique({
   146→   146→      where: { serviceId_key: { serviceId, key: STATS_KEY } },
   147→   147→      select: { valueJson: true },
   148→   148→    });
   149→   149→    if (!row) return;
   150→   150→    const saved = safeParse<Partial<LiveState['stats']> | null>(row.valueJson, null);
   151→   151→    if (saved && typeof saved.totalRequests === 'number') {
   152→   152→      const st = getLiveState(serviceId);
   153→   153→      st.stats = { ...emptyStats(), ...saved, byPath: saved.byPath ?? {} };
   154→   154→    }
   155→   155→  } catch {
   156→   156→    /* stats are best-effort */
   157→   157→  }
   158→   158→}

   160→   160→function scheduleStatsFlush(serviceId: string): void {
   161→   161→  const st = getLiveState(serviceId);
   162→   162→  if (st.flushTimer) return;
   163→   163→  st.flushTimer = setTimeout(() => {
   164→   164→    st.flushTimer = null;
   165→   165→    const snapshot = JSON.stringify(st.stats);
   166→   166→    db.serviceData
   167→   167→      .upsert({
   168→   168→        where: { serviceId_key: { serviceId, key: STATS_KEY } },
   169→   169→        update: { valueJson: snapshot },
   170→   170→        create: { serviceId, key: STATS_KEY, valueJson: snapshot },
   171→   171→      })
   172→   172→      .catch(() => {});
   173→   173→  }, 2500);
   174→   174→  st.flushTimer.unref?.();
   175→   175→}

   177→   177→function recordRequest(serviceId: string, entry: { method: string; path: string; status: number; ms: number }): void {
   178→   178→  const st = getLiveState(serviceId);
   179→   179→  st.stats.totalRequests += 1;
   180→   180→  if (entry.status < 400) st.stats.okResponses += 1;
   181→   181→  else st.stats.errorResponses += 1;
   182→   182→  st.stats.totalMs += entry.ms;
   183→   183→  st.stats.lastRequestAt = new Date().toISOString();
   184→   184→
   185→   185→  const bucket = st.stats.byPath[entry.path] ?? { count: 0, totalMs: 0 };
   186→   186→  bucket.count += 1;
   187→   187→  bucket.totalMs += entry.ms;
   188→   188→  st.stats.byPath[entry.path] = bucket;
   189→   189→
   190→   190→  st.ring.push({ ...entry, at: new Date().toISOString() });
   191→   191→  if (st.ring.length > RING_CAP) st.ring.splice(0, st.ring.length - RING_CAP);
   192→   192→
   193→   193→  st.latencies.push(entry.ms);
   194→   194→  if (st.latencies.length > LATENCY_CAP) st.latencies.splice(0, st.latencies.length - LATENCY_CAP);
   195→   195→
   196→   196→  scheduleStatsFlush(serviceId);
   197→   197→}

   199→   199→function percentile(values: number[], p: number): number {
   200→   200→  if (!values.length) return 0;
   201→   201→  const sorted = [...values].sort((a, b) => a - b);
   202→   202→  const idx = Math.min(sorted.length - 1, Math.ceil((p / 100) * sorted.length) - 1);
   203→   203→  return Math.round(sorted[Math.max(0, idx)]);
   204→   204→}

   662→   662→function readBody(req: http.IncomingMessage): Promise<Buffer> {
   663→   663→  return new Promise((resolve, reject) => {
   664→   664→    const chunks: Buffer[] = [];
   665→   665→    req.on('data', (c: Buffer) => {
   666→   666→      chunks.push(c);
   667→   667→      if (chunks.reduce((n, b) => n + b.length, 0) > MAX_VALUE_BYTES + 4096) {
   668→   668→        reject(new Error('payload too large'));
   669→   669→        req.destroy();
   670→   670→      }
   671→   671→    });
   672→   672→    req.on('end', () => resolve(Buffer.concat(chunks)));
   673→   673→    req.on('error', reject);
   674→   674→  });
   675→   675→}

   910→   910→async function startRuntime(svc: {
   911→   911→  id: string; name: string; type: string; commitHash: string; branch: string;
   912→   912→  hardwareTier: string; region: string; description: string; protocol: string;

   914→   914→  const meta: RunnerMeta = {

   916→   916→    commit: svc.commitHash,
   917→   917→    branch: svc.branch,

   919→   919→    region: svc.region,
   920→   920→    description: svc.description,
   921→   921→    protocol: svc.protocol,
   922→   922→  };
   923→   923→

   934→   934→    pid: process.pid,
   935→   935→    port,
   936→   936→    startedAt: new Date().toISOString(),
   937→   937→    mode: 'builtin-runner',
   938→   938→    healthy: true,
   939→   939→  };
   940→   940→  await db.service.update({ where: { id: svc.id }, data: { runtimeJson: JSON.stringify(state) } });
   941→   941→  await loadPersistedStats(svc.id);
   942→   942→  await writeLog(svc.id, `Runtime for "${svc.name}" is listening on 127.0.0.1:${port} (builtin runner v${RUNNER_VERSION}). Interactive console: /api/ingress/${svc.name}`);
   943→   943→  return { ...state, healthy: true };
   944→   944→}

   966→   966→async function replaceRuntime(svc: {
   967→   967→  id: string; name: string; status: string; type: string; commitHash: string; branch: string;
   968→   968→  hardwareTier: string; region: string; description: string; protocol: string;
   969→   969→}): Promise<ServiceRuntime | null> {
   970→   970→  const old = runners.get(svc.id);

   972→   972→    await new Promise<void>((resolve) => old.close(() => resolve()));

   974→   974→  }
   975→   975→  return startRuntime(svc);

   978→   978→/**
   979→   979→ * Reconcile one service with the desired state:
   980→   980→ *  - running  → guarantee a live listener with the CURRENT app version
   981→   981→ *  - anything else → guarantee no listener
   982→   982→ * Returns fresh runtime info for running services.
   983→   983→ */
   984→   984→export async function ensureRuntime(svc: {
   985→   985→  id: string; name: string; status: string; type: string; commitHash: string; branch: string;
   986→   986→  hardwareTier: string; region: string; description: string; protocol: string;
   987→   987→}): Promise<ServiceRuntime | null> {
   988→   988→  if (svc.status !== 'running') {
   989→   989→    await stopRuntime(svc);
   990→   990→    return null;
   991→   991→  }
   992→   992→
   993→   993→  const inProcess = runners.get(svc.id);
   994→   994→  if (inProcess) {
   995→   995→    const version = (inProcess as http.Server & { __v?: number }).__v;
   996→   996→    if (version !== RUNNER_VERSION) {
   997→   997→      return replaceRuntime(svc);
   998→   998→    }
   999→   999→    const state = await readRuntimeState(svc.id);
  1000→  1000→    if (state) {
  1001→  1001→      const healthy = await isHealthy(state.port);
  1002→  1002→      if (healthy !== state.healthy) {
  1003→  1003→        const updated = { ...state, healthy };
  1004→  1004→        await db.service.update({ where: { id: svc.id }, data: { runtimeJson: JSON.stringify(updated) } });
  1005→  1005→        return updated;
  1006→  1006→      }
  1007→  1007→      return { ...state, healthy };
  1008→  1008→    }
  1009→  1009→  }
  1010→  1010→
  1011→  1011→  // Not served by THIS process — maybe another worker/instance already serves it.
  1012→  1012→  const stored = await readRuntimeState(svc.id);
  1013→  1013→  if (stored) {
  1014→  1014→    const healthy = await isHealthy(stored.port);
  1015→  1015→    if (healthy) {
  1016→  1016→      if (!stored.healthy) {
  1017→  1017→        const updated = { ...stored, healthy: true };
  1018→  1018→        await db.service.update({ where: { id: svc.id }, data: { runtimeJson: JSON.stringify(updated) } });
  1019→  1019→        return updated;
  1020→  1020→      }
  1021→  1021→      return { ...stored, healthy: true };
  1022→  1022→    }
  1023→  1023→  }
  1024→  1024→
  1025→  1025→  const fresh = await startRuntime(svc);
  1026→  1026→  return fresh;
  1027→  1027→}

  1032→  1032→/** Reconciles all services (started missing runners, stops orphans). Cheap to call often. */
  1033→  1033→export async function reconcileRuntimes(): Promise<void> {
  1034→  1034→  rg.__hosterReconcile ??= { busy: false, lastRun: 0 };
  1035→  1035→  const rec = rg.__hosterReconcile;
  1036→  1036→  if (rec.busy || Date.now() - rec.lastRun < 2_000) return;
  1037→  1037→  rec.busy = true;
  1038→  1038→  try {
  1039→  1039→    const rows = await db.service.findMany({
  1040→  1040→      where: { OR: [{ status: 'running' }, { runtimeJson: { not: null } }] },
  1041→  1041→    });

  1047→  1047→      }
  1048→  1048→    }
  1049→  1049→  } finally {
  1050→  1050→    rec.busy = false;
  1051→  1051→    rec.lastRun = Date.now();
  1052→  1052→  }
  1053→  1053→}
